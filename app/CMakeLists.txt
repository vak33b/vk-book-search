cmake_minimum_required(VERSION 3.24)
project(vk_book_search VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Drogon CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc)
add_library(book_core src/core.cpp)
target_include_directories(book_core PUBLIC src)
target_link_libraries(book_core PUBLIC Drogon::Drogon ICU::uc SQLite3::SQLite3 CURL::libcurl)
target_compile_options(book_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vk_books src/main.cpp)
target_link_libraries(vk_books PRIVATE book_core)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE book_core)
add_test(NAME search_contract COMMAND core_tests)
